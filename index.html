const loginUser = asyncHandler(async (req, res) => {
  const { email, password } = req.validatedBody; // Joi validated
  const user = await User.findOne({ email }).select('+password');
  if (!user || !(await bcrypt.compare(password, user.password))) {
    throw new ApiError(401, 'Invalid credentials');
  }
  const token = jwt.sign({ id: user._id, role: user.role }, process.env.JWT_SECRET, { expiresIn: '24h' });
  return res.status(200).json({ success: true, token, role: user.role });
});

const getProduct = async (id) => {
  const key = `item:${id}`;
  const cached = await redisClient.get(key);
  if (cached) return { source: 'redis', data: JSON.parse(cached) };

  const product = await Product.findById(id).lean();
  if (!product) throw new ApiError(404, 'Item not found');
  await redisClient.setEx(key, 3600, JSON.stringify(product));
  return { source: 'mongodb', data: product };
};

// middlewares/verifyWebhook.js - Cryptographic Security
const verifyRazorpay = (req, res, next) => {
  const signature = req.headers['x-razorpay-signature'];
  const expected = crypto
    .createHmac('sha256', process.env.RAZORPAY_WEBHOOK_SECRET)
    .update(JSON.stringify(req.body))
    .digest('hex');
  if (signature !== expected) throw new ApiError(400, 'Invalid signature');
  next();
};

Technical Depth

Backend & Engineering Highlights

A practical look at how I implement core architectural patterns, integrations, and production safeguards.

REST API Architecture

Structured using a clean Router → Controller → Service → Model separation of concerns. Endpoints adhere strictly to HTTP standards (GET, POST, PUT, PATCH, DELETE) with predictable resource naming and standardized response envelopes (`{ success, data, message, error }`).

Clean Architecture

Authentication & Authorization

Implemented stateless authentication using JSON Web Tokens (JWT) with configurable expiry. Passwords are securely hashed with bcrypt using strong salt rounds. Route-level middleware enforces Role-Based Access Control (RBAC) to guard administrative endpoints.

JWT • bcrypt • RBAC

MongoDB & Mongoose

Designed flexible document schemas with strict Mongoose validation, sub-document relationships, and pre-save hooks. Implemented indexing on high-frequency query fields to prevent full-collection scans and utilized aggregation pipelines for analytics queries.

Schema Design • Indexing

Redis Caching

Implemented the Cache-Aside pattern using Redis for fast-access reads, atomic counters, and TTL-based expiration. Significantly reduced repeated expensive database queries and enabled sliding-window rate limiting to protect critical endpoints.

Cache-Aside • TTL • Counters

Payment Integration (Razorpay)

Integrated the complete Razorpay payment lifecycle: server-side order generation, client-side checkout handling, payment capture verification via cryptographic signature checks, and refund API handling with transactional status updates.

Razorpay • Orders • Refunds

Webhooks & Event Handling

Engineered secure webhook listeners that verify incoming payload integrity using HMAC SHA-256 signatures. Designed idempotent processing logic to ensure identical webhook delivery retries do not duplicate orders or financial transactions.

HMAC Verification • Idempotency

API Validation (Joi)

Enforced strict request schema validation at the router boundary using Joi middleware. Rejects malformed bodies, unexpected types, or missing required fields early with a descriptive 400 Bad Request before reaching core business logic.

Joi • Fail Fast • Sanitization

Centralized Error Handling

Architected custom Express error middleware (err, req, res, next). Distinguishes between operational errors (4xx client mistakes) and internal programmer errors (500), sanitizes stack traces in production, and maintains uniform JSON error structures.

Express Middleware • 4xx/5xx

Cron Jobs & Maintenance

Implemented automated background schedules using node-cron for recurring maintenance: archiving expired guest sessions, cleaning temporary upload files, and triggering periodic health syncs without blocking the Node.js event loop.

node-cron • Non-blocking Tasks

File Uploads (Multer & Cloudinary)

Configured multipart upload streams using Multer with memory/disk buffers, mime-type filtering, and file-size constraints. Uploaded assets directly to Cloudinary CDN with automated format optimization and secure remote URL storage.

Multer • Cloudinary • CDN

Containerization with Docker

Packaged applications into lightweight Docker containers using multi-stage builds. Ensured reproducible development environments, isolated dependencies, and standardized runtime configuration across local setups and staging deployments.

Docker • Multi-Stage • Portability

API Documentation & Postman

Maintained comprehensive Swagger / OpenAPI specifications for interactive endpoint exploration and published curated Postman test collections with automated assertion scripts for regression verification.

Swagger • Postman Collections

Production Node.js Clean Architecture Layout

Separation of Concerns 
src/
├── config/         # Joi-validated env, DB connection pools (MongoDB/MySQL), Redis client
├── controllers/    # Pure HTTP handlers: status codes, response envelopes, zero direct queries
├── middlewares/    # auth(JWT+RBAC), validate(Joi), rateLimiter, centralized errorHandler
├── models/         # Mongoose schemas with compound indexes / Sequelize relational models
├── routes/         # Semantic versioned endpoints (e.g. /api/v1/orders)
├── services/       # Decoupled business rules, third-party APIs (Razorpay), transactions
├── utils/          # ApiError, ApiResponse, logger, asyncHandler
└── app.js          # Express server setup, CORS, Helmet, graceful shutdown handling
Fail-Fast Boundary Validation

Every payload is inspected by Joi middleware before entering controller functions; invalid inputs return 400 instantly.

Zero Database Leaks in HTTP Layer

Controllers never query databases directly. Data access is encapsulated in Services/Models for easy unit testing and mocking.

Graceful Process Lifecycle

Handlers for SIGTERM and SIGINT cleanly close database pools and finish in-flight requests before exiting.
